rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isSelf(uid) { return signedIn() && request.auth.uid == uid; }
    function isDeveloper() { return signedIn() && request.auth.token.email == 'dev@wbsquantum.app'; }

    match /users/{userId} {
      allow read, write: if isSelf(userId) || isDeveloper();

      match /tasks/{taskId} {
        allow read, create, update, delete: if isSelf(userId) || isDeveloper();
      }
      match /exams/{examId} {
        allow read, create, update, delete: if isSelf(userId) || isDeveloper();
      }
      match /notes/{noteId} {
        allow read, create, update, delete: if isSelf(userId) || isDeveloper();
      }
      match /preferences/{document} {
        allow read, write: if isSelf(userId) || isDeveloper();
      }
      match /revisionSessions/{sessionId} {
        allow read, create, update, delete: if isSelf(userId) || isDeveloper();
      }
    }

    match /appConfig/maintenance {
      allow read: if signedIn();
      allow create, update, delete: if isDeveloper();
    }

    match /appConfig/features {
      allow read: if signedIn();
      allow create, update, delete: if isDeveloper();
    }

    match /streaks/{userId} {
      allow read: if signedIn();
      allow create, update, delete: if isSelf(userId);
    }

    match /leaderboard/{userId} {
      allow read: if signedIn();
      allow create, update, delete: if isSelf(userId) || isDeveloper();
    }
  }
}
